std::string getEmail(char *buff)
{
    std::string tmp = buff;
    std::stringstream ss(tmp);
    std::string email;

    std::getline(ss, email, '&');
    cout << " --------------> " << email << std::endl;
    return (email);
}

int authenCheck(Record &log, Request &req, SessionsList &sList)
{
    bool logged = true;
    if (log.data.loginToken == req.path && logged)
    {
        if (req.method != "GET")
            return 0;
        log.data.retPath = log.data.redirectTokens[0];
        log.data.retValid = true;
        log.data.retCode = 302;
        return 1;
    }
    else
    {
        if (req.method == "DELETE")
            return 0;
        for (size_t i = 0; i < log.data.redirectTokens.size(); i++)
        {
            if (log.data.redirectTokens[i] == req.path && req.method == "POST")
            {
                char *buff = NULL;
                int fd = open(req.bodyPath.c_str(), O_RDONLY);
                if (fd == -1) return (1);
                int bytes = read(fd, buff, SIZESR);
                (void) bytes;
                if (bytes == -1) return (1);
                std::string email = getEmail(buff);
                sList.storeInputs(req, email);
            }
            if (log.data.redirectTokens[i] == req.path && !logged)
            {
                log.data.retPath = log.data.loginToken;
                log.data.retValid = true;
                log.data.retCode = 302;
                return 1;
            }
            else if (log.data.redirectTokens[i] == req.path && logged)
            {
                if (req.method == "POST")
                    req.method = "GET";
                return 0;
            }
        }
    }
    return 0;
}
